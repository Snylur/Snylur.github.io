from time import sleep
def roll_credits(credits, pause=0.5, extra_pause=1.5) -> bool: 
  """
  Display `credits` line by line, with `pause` seconds
  of pause between each line, and `extra_pause` seconds
  of pause at section break (a line containing only `=`
  characters). Returns boolean status. 
  """
  try: 
    for line in credits.split('\n'): 
      print(line)
      sleep(extra_pause if line == '=' * len(line) else pause)
  except BaseException as e: 
    print("[red]Error raised in [purple]roll_credits[/]. [/]")
    return False
  else: 
    return True

# helper.py

class User: 
  def __init__(self, name, age=0): 
    self.__name, self.__age = name, age
  def set_age(self, age=None): 
    old = self.__age
    if age is None: age = old + 1
    self.__age = age
    return old
  def get_name(self): return self.__name
  def get_age(self): return self.__age

class World:
  def __init__(self, name, users=None):
    if users is None: users = []
    self.__name, self.__users = name, users
  def register_user(self, *args, **kwargs): 
    self.__users.append(User(*args, **kwargs))
  def get_users_by_name(self, name): 
    return [
      user for user in self.__users
      if user.get_name() == name
    ]
